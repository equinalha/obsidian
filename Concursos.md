TABLE
  file.link AS "Prova",
  Desempenho,
  Banca,
  Obs,
  row["1o Colocado"] AS "1º Colocado",
  Tipo,
  Obj,
  Status,
  Data,
  round(((Desempenho / row["1o Colocado"]) - 1) * 100) + "%" AS "% Colocação"
FROM ""
WHERE base = link("Simulados.base")
SORT file.name ASC
